/*
 * Module 8: kampf_animation.js
 * Dependencies: shared Babylon scene, player / weapon services.
 * Public service: combat.
 *
 * Ragdoll requires an explicitly registered physics adapter.
 * Without one, incapacitated characters use a controlled fallback state.
 */

const COMBAT_ACTIONS = Object.freeze({
  melee: {
    windup: 0.19,
    active: 0.12,
    recovery: 0.34,
    rangeM: 1.55,
    halfAngleRad: Math.PI * 0.34,
    damage: 18,
    knockbackMps: 2.7,
    staminaCost: 7
  },
  heavyMelee: {
    windup: 0.39,
    active: 0.18,
    recovery: 0.58,
    rangeM: 1.8,
    halfAngleRad: Math.PI * 0.30,
    damage: 32,
    knockbackMps: 5.2,
    staminaCost: 17
  },
  takedown: {
    windup: 0.3,
    active: 0.3,
    recovery: 0.8,
    rangeM: 1.25,
    halfAngleRad: Math.PI * 0.3,
    damage: 100,
    knockbackMps: 0,
    staminaCost: 20
  }
});

const COMBAT_HIT_REACTIONS = Object.freeze({
  head: "headHit",
  torso: "bodyHit",
  arm: "armHit",
  leg: "legHit",
  default: "bodyHit"
});

function combatClamp(value, min, max) {
  return Math.max(min, Math.min(max, Number.isFinite(value) ? value : min));
}

function combatFinite(value, fallback = 0) {
  return Number.isFinite(value) ? value : fallback;
}

function combatSmoothstep(t) {
  const v = combatClamp(t, 0, 1);
  return v * v * (3 - 2 * v);
}

export async function createCombatAnimationSystem({
  runtime,
  scene,
  camera,
  services,
  entities,
  logger
}) {
  if (!runtime || !scene || !camera) {
    throw new TypeError("Combat requires the shared runtime, scene and camera.");
  }

  const B = globalThis.BABYLON;
  if (!B) throw new Error("Babylon.js is unavailable.");

  const eventDisposers = [];
  const serviceDisposers = [];
  const combatants = new Map();
  const activeAttacks = new Map();
  const ragdollHandles = new Map();
  const carriedBodies = new Map();
  const coverStates = new Map();

  let nextAttackId = 1;
  let elapsed = 0;
  let initialized = false;
  let running = false;
  let disposed = false;

  const replayConfig = {
    sampleInterval: 0.08,
    maxSeconds: 24,
    samples: [],
    events: [],
    sampleAccumulator: 0,
    replay: null,
    recording: true,
    maximumEventEntries: 240
  };

  const hitProxyMaterial = new B.StandardMaterial(
    "combat-hit-proxy-material",
    scene
  );
  hitProxyMaterial.diffuseColor = new B.Color3(1, 1, 1);
  hitProxyMaterial.alpha = 0;
  hitProxyMaterial.disableLighting = true;
  hitProxyMaterial.backFaceCulling = false;

  const LOOPING_ANIMATION_STATES = new Set([
    "idle", "walk", "run", "sprint", "cover", "aim", "carry",
    "crouch", "crawl", "swim", "dive", "vehicleInteraction"
  ]);

  function getPlayerService() {
    return services.get("player");
  }

  function characterNode(id) {
    return scene.getTransformNodeByName(`character-root-${id}`) ??
      scene.getMeshByName(`character-root-${id}`) ??
      null;
  }

  function entityPosition(record) {
    if (typeof record.getPosition === "function") {
      const position = record.getPosition();
      if (position) {
        return position.clone
          ? position.clone()
          : new B.Vector3(position.x, position.y, position.z);
      }
    }

    if (record.root?.getAbsolutePosition) {
      return record.root.getAbsolutePosition().clone();
    }

    if (record.mesh?.getAbsolutePosition) {
      return record.mesh.getAbsolutePosition().clone();
    }

    if (record.position) {
      return record.position.clone
        ? record.position.clone()
        : new B.Vector3(
            record.position.x,
            record.position.y,
            record.position.z
          );
    }

    return null;
  }

  function entityForward(record) {
    if (record.root) {
      const direction = record.root.forward?.clone();
      if (direction && direction.lengthSquared() > 0.001) {
        direction.y = 0;
        return direction.normalize();
      }
    }

    const heading = combatFinite(record.heading, 0);

    return new B.Vector3(
      Math.sin(heading),
      0,
      Math.cos(heading)
    );
  }

  function createHitboxProxies(record) {
    if (
      !record?.root ||
      record.root.isDisposed?.() ||
      record.hitboxMeshes.length
    ) {
      return;
    }

    // Transparent, pickable hit zones make procedural and imported characters
    // available to the same projectile raycast path without relying on the
    // render meshes themselves being pickable.
    const zones = [
      { zone: "head", position: [0, 1.78, 0.01], scale: [0.145, 0.18, 0.14] },
      { zone: "torso", position: [0, 1.20, 0], scale: [0.23, 0.36, 0.17] },
      { zone: "arm", position: [-0.29, 1.22, 0], scale: [0.11, 0.31, 0.12] },
      { zone: "arm", position: [0.29, 1.22, 0], scale: [0.11, 0.31, 0.12] },
      { zone: "leg", position: [-0.12, 0.49, 0], scale: [0.12, 0.32, 0.13] },
      { zone: "leg", position: [0.12, 0.49, 0], scale: [0.12, 0.32, 0.13] }
    ];

    for (let index = 0; index < zones.length; index++) {
      const zone = zones[index];
      const mesh = B.MeshBuilder.CreateSphere(
        `combat-hitbox-${record.id}-${zone.zone}-${index}`,
        { diameter: 1, segments: 8 },
        scene
      );

      mesh.parent = record.root;
      mesh.position.set(
        zone.position[0],
        zone.position[1],
        zone.position[2]
      );
      mesh.scaling.set(
        zone.scale[0],
        zone.scale[1],
        zone.scale[2]
      );
      mesh.material = hitProxyMaterial;
      mesh.visibility = 0;
      mesh.isVisible = true;
      mesh.isPickable = true;
      mesh.checkCollisions = false;
      mesh.metadata = {
        ...(mesh.metadata || {}),
        combatantId: record.id,
        characterId: record.id,
        hitZone: zone.zone,
        isCombatHitbox: true,
        blocksPlayer: false,
        projectilePassThrough: false,
        materialClass: "flesh",
        materialThicknessM: 0.15,
        penetrationResistanceJPerM: 3000,
        penetrable: true
      };

      record.hitboxMeshes.push(mesh);
    }
  }

  function disposeHitboxProxies(record) {
    for (const mesh of record?.hitboxMeshes ?? []) {
      try {
        mesh.dispose(false, false);
      } catch {}
    }

    if (record?.hitboxMeshes) record.hitboxMeshes.length = 0;
  }

  function syncPlayerCombatState(record) {
    if (!record || record.type !== "player") return;

    const profile = getPlayerService()?.getCharacter?.(record.id);
    if (!profile) return;

    if (Number.isFinite(profile.health)) {
      const wasDead = record.dead;
      record.health = combatClamp(profile.health, 0, record.maxHealth);
      record.dead = record.health <= 0;

      if (record.dead) {
        record.incapacitated = true;
        record.state = "dead";
      } else if (wasDead) {
        record.incapacitated = false;
        record.state = "neutral";
        record.ragdollEnabled = false;
        try {
          ragdollHandles.get(record.id)?.dispose?.();
        } catch {}
        ragdollHandles.delete(record.id);
        record.root?.setEnabled(true);
      }
    }

    if (Number.isFinite(profile.stamina)) {
      record.stamina = combatClamp(profile.stamina, 0, 100);
    }
  }

  function recordReplayEvent(name, payload = {}) {
    if (!replayConfig.recording) return;

    replayConfig.events.push({
      t: elapsed,
      name,
      payload: {
        targetId: payload.targetId ?? null,
        attackerId: payload.attackerId ?? payload.shooterId ?? null,
        characterId: payload.characterId ?? null,
        vehicleId: payload.vehicleId ?? null,
        missionId: payload.missionId ?? null,
        position: (() => {
          const point = payload.position ??
            payload.muzzlePosition ??
            payload.hitPosition;

          return point && Number.isFinite(point.x)
            ? { x: point.x, y: point.y, z: point.z }
            : null;
        })()
      }
    });

    if (replayConfig.events.length > replayConfig.maximumEventEntries) {
      replayConfig.events.splice(
        0,
        replayConfig.events.length - replayConfig.maximumEventEntries
      );
    }
  }

  function updateAnimationBlend(record, dt) {
    const blend = record?.animationBlendState;
    if (!blend) return;

    blend.elapsed = Math.min(blend.duration, blend.elapsed + dt);
    const t = combatSmoothstep(blend.elapsed / blend.duration);

    try {
      blend.from?.setWeightForAllAnimatables?.(1 - t);
      blend.to?.setWeightForAllAnimatables?.(t);
    } catch (error) {
      logger.warn("Animation blend update failed.", error);
    }

    if (blend.elapsed >= blend.duration) {
      if (blend.from && blend.from !== blend.to) {
        try {
          blend.from.stop();
        } catch {}
      }

      blend.to?.setWeightForAllAnimatables?.(1);
      record.activeAnimationGroup = blend.to;
      record.animationBlendState = null;
    }
  }

  function makeCombatant(definition) {
    if (!definition?.id) {
      throw new TypeError("A combatant requires a stable ID.");
    }

    if (combatants.has(definition.id)) {
      throw new Error(`Combatant already registered: ${definition.id}`);
    }

    const maximumHealth = Math.max(
      1,
      combatFinite(definition.maxHealth, 100)
    );

    const record = {
      id: definition.id,
      type: definition.type ?? "npc",
      faction: definition.faction ?? "civilian",

      root: definition.root ?? null,
      mesh: definition.mesh ?? null,
      getPosition: definition.getPosition ?? null,
      position: definition.position ?? null,
      heading: combatFinite(definition.heading, 0),
      animationGroups: [...(definition.animationGroups ?? [])],
      activeAnimationGroup: null,
      animationBlendState: null,
      hitboxMeshes: [],
      bodyMassKg: combatClamp(
        combatFinite(definition.bodyMassKg ?? definition.metadata?.massKg, 75),
        1,
        500
      ),

      health: combatClamp(
        combatFinite(definition.health, maximumHealth),
        0,
        maximumHealth
      ),

      maxHealth: maximumHealth,
      stamina: combatClamp(
        combatFinite(definition.stamina, 100),
        0,
        100
      ),

      armor: combatClamp(
        combatFinite(definition.armor, 0),
        0,
        100
      ),

      state: "neutral",
      hitReaction: null,
      incapacitated: false,
      dead: false,

      targetId: null,
      coverId: null,
      hostageTargetId: null,
      carriedBy: null,

      cooldownUntil: 0,
      lastDamageAt: -Infinity,
      lastAttackerId: null,

      hitZones: definition.hitZones ?? null,
      ragdollEnabled: false,
      defense: {
        blocking: false,
        dodgingUntil: -Infinity,
        blockDamageMultiplier: 0.32
      },

      metadata: { ...(definition.metadata ?? {}) }
    };

    combatants.set(record.id, record);
    createHitboxProxies(record);

    return record;
  }

  function registerCombatant(definition) {
    const record = makeCombatant(definition);

    runtime.events.emit("combat:combatantRegistered", {
      combatantId: record.id,
      type: record.type,
      faction: record.faction
    });

    return Object.freeze({
      id: record.id,
      get: () => combatants.get(record.id) ?? null,
      dispose: () => unregisterCombatant(record.id)
    });
  }

  function unregisterCombatant(id) {
    const record = combatants.get(id);
    if (!record) return false;

    cancelAttack(id, "combatant-unregistered");
    releaseFromCover(id);
    releaseCarriedBody(id);

    const ragdoll = ragdollHandles.get(id);

    try {
      ragdoll?.dispose?.();
    } catch (error) {
      logger.warn("Ragdoll cleanup failed.", error);
    }

    ragdollHandles.delete(id);
    disposeHitboxProxies(record);
    combatants.delete(id);

    runtime.events.emit("combat:combatantUnregistered", {
      combatantId: id
    });

    return true;
  }

  function getCombatant(id) {
    return combatants.get(id) ?? null;
  }

  function registerDefaultPlayerCharacters() {
    const player = getPlayerService();

    for (const id of ["lucia", "jason"]) {
      const profile = player?.getCharacter?.(id);
      if (!profile) continue;

      const root = characterNode(id);
      const existing = combatants.get(id);

      if (existing) {
        existing.root = root ?? existing.root;
        existing.position = profile.position;
        existing.getPosition = () => profile.position;
        existing.health = combatClamp(
          combatFinite(profile.health, existing.health),
          0,
          existing.maxHealth
        );
        existing.stamina = combatClamp(
          combatFinite(profile.stamina, existing.stamina),
          0,
          100
        );

        if (existing.root && !existing.hitboxMeshes.length) {
          createHitboxProxies(existing);
        }
        continue;
      }

      registerCombatant({
        id,
        type: "player",
        faction: "player",
        root,
        position: profile.position,
        getPosition: () => profile.position,
        maxHealth: 100,
        health: profile.health,
        stamina: profile.stamina,
        metadata: { protagonist: true }
      });
    }
  }

  /* --------------------------------------------------------------- */
  /* Attack lifecycle                                                  */
  /* --------------------------------------------------------------- */

  function beginAttack(attackerId, attackType = "melee", options = {}) {
    const attacker = getCombatant(attackerId);
    const config = COMBAT_ACTIONS[attackType];

    if (!attacker || !config) {
      return { ok: false, reason: "invalid-attacker-or-action" };
    }

    if (attacker.dead || attacker.incapacitated) {
      return { ok: false, reason: "attacker-incapacitated" };
    }

    if (activeAttacks.has(attackerId)) {
      return { ok: false, reason: "attack-already-active" };
    }

    if (elapsed < attacker.cooldownUntil) {
      return { ok: false, reason: "attack-on-cooldown" };
    }

    if (attacker.stamina < config.staminaCost) {
      return { ok: false, reason: "insufficient-stamina" };
    }

    attacker.stamina -= config.staminaCost;
    const playerProfile = getPlayerService()?.getCharacter?.(attackerId);
    if (playerProfile) playerProfile.stamina = attacker.stamina;

    const attack = {
      id: `combat-attack-${nextAttackId++}`,
      attackerId,
      targetId: options.targetId ?? null,
      type: attackType,
      config,
      phase: "windup",
      phaseElapsed: 0,
      hitTargets: new Set(),
      startedAt: elapsed,
      metadata: {
        weaponId: options.weaponId ?? null,
        missionId: options.missionId ?? null
      }
    };

    activeAttacks.set(attackerId, attack);
    attacker.state = attackType === "takedown"
      ? "takedown"
      : "attacking";

    playAnimation(attacker, attackType);

    runtime.events.emit("combat:attackStarted", {
      attackId: attack.id,
      attackerId,
      type: attackType,
      targetId: attack.targetId
    });

    return { ok: true, attackId: attack.id };
  }

  function cancelAttack(attackerId, reason = "cancelled") {
    const attack = activeAttacks.get(attackerId);
    if (!attack) return false;

    activeAttacks.delete(attackerId);

    const attacker = getCombatant(attackerId);

    if (attacker && !attacker.dead && !attacker.incapacitated) {
      attacker.state = "neutral";
    }

    runtime.events.emit("combat:attackCancelled", {
      attackId: attack.id,
      attackerId,
      reason
    });

    return true;
  }

  function canTarget(
    attacker,
    target,
    maxRange,
    halfAngleRad = COMBAT_ACTIONS.melee.halfAngleRad
  ) {
    if (!attacker || !target) return false;
    if (target.id === attacker.id) return false;
    if (target.dead || target.incapacitated || target.carriedBy) return false;
    if (target.metadata.invulnerable === true) return false;

    // Friendly-fire policy is explicit per combatant.
    if (
      attacker.faction === target.faction &&
      attacker.metadata.friendlyFire !== true
    ) {
      return false;
    }

    const a = entityPosition(attacker);
    const b = entityPosition(target);

    if (!a || !b) return false;

    const toTarget = b.subtract(a);
    toTarget.y = 0;

    const distance = toTarget.length();
    if (distance > maxRange || distance < 0.05) return false;

    const forward = entityForward(attacker);
    const dot = B.Vector3.Dot(forward, toTarget.normalize());

    return dot >= Math.cos(halfAngleRad);
  }

  function hasLineOfSight(from, to, targetId) {
    const delta = to.subtract(from);
    const distance = delta.length();

    if (distance < 0.02) return true;

    const ray = new B.Ray(
      from,
      delta.normalize(),
      Math.max(0.01, distance - 0.12)
    );

    const hit = scene.pickWithRay(
      ray,
      mesh => {
        if (!mesh || mesh.isDisposed()) return false;
        if (!mesh.isEnabled() || !mesh.isVisible) return false;
        if (mesh.metadata?.blocksPlayer === false) return false;
        if (mesh.metadata?.isWaterSurface) return false;
        if (mesh.metadata?.combatantId === targetId) return false;
        if (mesh.metadata?.characterId === targetId) return false;
        return mesh.isPickable !== false;
      },
      false
    );

    return !hit?.hit;
  }

  function performAttackHit(attack) {
    const attacker = getCombatant(attack.attackerId);
    if (!attacker) return;

    const origin = entityPosition(attacker);
    if (!origin) return;

    origin.y += 1.0;

    const forward = entityForward(attacker);
    const candidates = [];

    if (attack.targetId) {
      const target = getCombatant(attack.targetId);
      if (target) candidates.push(target);
    } else {
      for (const target of combatants.values()) {
        if (canTarget(attacker, target, attack.config.rangeM, attack.config.halfAngleRad)) {
          candidates.push(target);
        }
      }
    }

    for (const target of candidates) {
      if (attack.hitTargets.has(target.id)) continue;

      if (!canTarget(attacker, target, attack.config.rangeM, attack.config.halfAngleRad)) continue;

      const targetPosition = entityPosition(target);
      if (!targetPosition) continue;

      const distance = B.Vector3.Distance(origin, targetPosition);
      const targetPoint = targetPosition.add(new B.Vector3(0, 0.95, 0));

      if (!hasLineOfSight(origin, targetPoint, target.id)) continue;

      // For takedowns, the target must already be vulnerable and close.
      if (
        attack.type === "takedown" &&
        target.state !== "stunned" &&
        target.state !== "surrendering" &&
        target.metadata.takedownVulnerable !== true
      ) {
        continue;
      }

      attack.hitTargets.add(target.id);

      applyDamage({
        targetId: target.id,
        attackerId: attacker.id,
        damage: attack.config.damage,
        hitZone: target.metadata.preferredHitZone ?? "torso",
        source: attack.type,
        impulse: forward.scale(attack.config.knockbackMps),
        attackId: attack.id
      });

      runtime.events.emit("combat:meleeHit", {
        attackId: attack.id,
        attackerId: attacker.id,
        targetId: target.id,
        distanceM: distance,
        attackType: attack.type
      });
    }
  }

  /* --------------------------------------------------------------- */
  /* Damage, hit reaction and ragdoll adapter                          */
  /* --------------------------------------------------------------- */

  function applyDamage({
    targetId,
    attackerId = null,
    damage,
    hitZone = "torso",
    source = "unknown",
    impulse = null,
    attackId = null
  }) {
    const target = getCombatant(targetId);

    if (!target) {
      return { ok: false, reason: "target-not-registered" };
    }

    if (target.dead || target.metadata.invulnerable === true) {
      return { ok: false, reason: "target-unavailable" };
    }

    syncPlayerCombatState(target);
    if (target.dead || target.metadata.invulnerable === true) {
      return { ok: false, reason: "target-unavailable" };
    }

    const incomingSource = String(source ?? "unknown");

    if (target.defense && elapsed < target.defense.dodgingUntil) {
      runtime.events.emit("combat:evaded", {
        targetId,
        attackerId,
        source: incomingSource,
        attackId,
        position: entityPosition(target)
      });
      return {
        ok: true,
        targetId,
        damage: 0,
        avoided: true,
        reason: "dodge-window"
      };
    }

    const amount = combatClamp(combatFinite(damage), 0, 10000);
    const isBallistic = incomingSource === "ballistic-projectile";
    const zoneMultiplier = isBallistic
      ? (hitZone === "head" ? 1.9 : hitZone === "arm" ? 0.72 : hitZone === "leg" ? 0.78 : 1)
      : 1;

    const zoneAdjustedDamage = amount * zoneMultiplier;
    const armorReduction = combatClamp(target.armor / 100, 0, 0.8);
    let preMitigationDamage = zoneAdjustedDamage;
    let blocked = false;

    if (
      target.defense?.blocking &&
      ["melee", "heavyMelee", "takedown"].includes(incomingSource)
    ) {
      const attacker = attackerId ? getCombatant(attackerId) : null;
      const targetPosition = entityPosition(target);
      const attackerPosition = attacker ? entityPosition(attacker) : null;

      if (targetPosition && attackerPosition) {
        const towardAttacker = attackerPosition.subtract(targetPosition);
        towardAttacker.y = 0;
        const forward = entityForward(target);

        if (
          towardAttacker.lengthSquared() > 1e-5 &&
          B.Vector3.Dot(forward, towardAttacker.normalize()) > 0.1
        ) {
          preMitigationDamage *= target.defense.blockDamageMultiplier;
          blocked = true;
        }
      }
    }

    const effectiveDamage = Math.max(
      0,
      preMitigationDamage * (1 - armorReduction)
    );

    const playerCharacter = getPlayerService()?.getCharacter?.(targetId);
    const startingHealth = playerCharacter
      ? combatClamp(
          combatFinite(playerCharacter.health, target.health),
          0,
          target.maxHealth
        )
      : target.health;

    target.health = Math.max(0, startingHealth - effectiveDamage);
    target.lastDamageAt = elapsed;
    target.lastAttackerId = attackerId;
    target.hitReaction = COMBAT_HIT_REACTIONS[hitZone] ??
      COMBAT_HIT_REACTIONS.default;

    if (playerCharacter) {
      playerCharacter.health = target.health;
    }

    if (blocked) {
      runtime.events.emit("combat:blocked", {
        targetId,
        attackerId,
        incomingDamage: amount,
        appliedDamage: effectiveDamage,
        source: incomingSource
      });
    }

    if (
      isBallistic &&
      (hitZone === "leg" || hitZone === "arm") &&
      effectiveDamage >= 8
    ) {
      const impairment = hitZone === "leg" ? "movement" : "weapon-handling";
      const key = hitZone === "leg" ? "movementPenalty" : "weaponHandlingPenalty";

      target.metadata[key] = combatClamp(
        effectiveDamage / Math.max(1, target.maxHealth * 0.55),
        0.08,
        0.65
      );

      runtime.events.emit("combat:limbInjured", {
        targetId,
        attackerId,
        hitZone,
        impairment,
        severity: target.metadata[key]
      });
    }

    if (
      isBallistic &&
      hitZone === "arm" &&
      effectiveDamage >= 20 &&
      !target.dead
    ) {
      runtime.events.emit("combat:disarmOpportunity", {
        targetId,
        attackerId,
        hitZone,
        weaponId: target.metadata.currentWeaponId ?? null
      });
    }

    const hitPosition = entityPosition(target);

    if (hitPosition && impulse) {
      const adapter = services.get("physics");
      const ragdoll = ragdollHandles.get(targetId);

      if (ragdoll?.applyImpulse && target.ragdollEnabled) {
        ragdoll.applyImpulse(hitPosition, impulse);
      } else if (adapter?.applyCharacterImpulse) {
        adapter.applyCharacterImpulse(targetId, hitPosition, impulse);
      }
    }

    runtime.events.emit("combat:hit", {
      targetId,
      attackerId,
      attackId,
      damage: effectiveDamage,
      hitZone,
      source,
      healthRemaining: target.health,
      position: hitPosition
    });

    if (target.health <= 0) {
      target.dead = true;
      target.incapacitated = true;
      target.state = "dead";

      enableRagdoll(targetId, {
        reason: "lethal-damage",
        impulse
      });

      runtime.events.emit("combat:knockout", {
        targetId,
        attackerId,
        lethal: true,
        position: hitPosition
      });

      runtime.events.emit("combat:death", {
        targetId,
        attackerId,
        source,
        position: hitPosition
      });
    } else if (
      effectiveDamage >= 22 ||
      hitZone === "head" ||
      hitZone === "leg"
    ) {
      target.state = "stunned";

      runtime.events.emit("combat:hitReaction", {
        targetId,
        hitZone,
        reaction: target.hitReaction,
        duration: hitZone === "head" ? 0.8 : 0.38
      });
    }

    return {
      ok: true,
      targetId,
      damage: effectiveDamage,
      healthRemaining: target.health
    };
  }

  function enableRagdoll(id, options = {}) {
    const record = getCombatant(id);
    if (!record) return { ok: false, reason: "unknown-combatant" };

    const physics = services.get("physics");

    if (typeof physics?.createCharacterRagdoll !== "function") {
      // This is a fallback incapacitation state, not a simulated ragdoll.
      record.ragdollEnabled = false;
      record.state = record.dead ? "dead" : "incapacitated";

      runtime.events.emit("combat:ragdollUnavailable", {
        combatantId: id
      });

      return {
        ok: false,
        reason: "ragdoll-physics-adapter-unavailable"
      };
    }

    try {
      const handle = physics.createCharacterRagdoll({
        characterId: id,
        root: record.root,
        mesh: record.mesh,
        impulse: options.impulse ?? null,
        reason: options.reason ?? "impact"
      });

      if (!handle || typeof handle.dispose !== "function") {
        throw new TypeError(
          "Physics adapter must return a disposable ragdoll handle."
        );
      }

      ragdollHandles.get(id)?.dispose?.();
      ragdollHandles.set(id, handle);
      record.ragdollEnabled = true;
      record.state = "ragdoll";

      return { ok: true };
    } catch (error) {
      logger.error("Ragdoll creation failed.", error);

      record.ragdollEnabled = false;
      record.state = "incapacitated";

      return {
        ok: false,
        reason: "ragdoll-creation-failed",
        error: error.message
      };
    }
  }

  function setDefenseState(id, state = {}) {
    const record = getCombatant(id);
    if (!record || record.dead || record.incapacitated) {
      return { ok: false, reason: "character-unavailable" };
    }

    if (state.blocking !== undefined) {
      record.defense.blocking = state.blocking === true;
    }

    if (state.dodging === true) {
      const duration = combatClamp(
        combatFinite(state.durationSeconds, 0.32),
        0.08,
        0.8
      );

      record.defense.dodgingUntil = Math.max(
        record.defense.dodgingUntil,
        elapsed + duration
      );
      record.defense.blocking = false;
    }

    if (state.dodging === false) record.defense.dodgingUntil = -Infinity;

    if (state.blockDamageMultiplier !== undefined) {
      record.defense.blockDamageMultiplier = combatClamp(
        combatFinite(state.blockDamageMultiplier, 0.32),
        0.05,
        1
      );
    }

    if (record.defense.blocking) playAnimation(record, "block");
    else if (state.dodging === true) playAnimation(record, "dodge");

    runtime.events.emit("combat:defenseStateChanged", {
      characterId: id,
      blocking: record.defense.blocking,
      dodging: elapsed < record.defense.dodgingUntil,
      dodgingUntil: record.defense.dodgingUntil
    });

    return {
      ok: true,
      blocking: record.defense.blocking,
      dodgingUntil: record.defense.dodgingUntil
    };
  }

  function recoverFromRagdoll(id) {
    const record = getCombatant(id);
    if (!record || record.dead) return false;

    const handle = ragdollHandles.get(id);

    try {
      handle?.disable?.();
      handle?.dispose?.();
    } catch (error) {
      logger.warn("Ragdoll recovery failed.", error);
    }

    ragdollHandles.delete(id);
    record.ragdollEnabled = false;
    record.incapacitated = false;
    record.state = "neutral";

    runtime.events.emit("combat:ragdollRecovered", {
      combatantId: id
    });

    return true;
  }

    walk: ["walk", "walking"],
    run: ["run", "jog", "running"],
    sprint: ["sprint", "sprinting"],
    melee: ["melee", "punch", "attack"],
    heavyMelee: ["heavy", "strong", "attack_heavy"],
    takedown: ["takedown", "finisher"],
    cover: ["cover", "crouch_cover"],
    aim: ["aim", "weapon_aim"],

    climb: ["climb", "climbing"],
    crouch: ["crouch", "crouching"],
    crawl: ["crawl", "crawling", "prone"],
    swim: ["swim", "swimming"],

    standUp: ["standup", "get_up", "rise"]
  });

  function playAnimation(record, semanticState, options = {}) {
    const groups = record.animationGroups;

    if (!groups.length) {
      runtime.events.emit("combat:animationRequested", {
        combatantId: record.id,
        state: semanticState,
        blendSeconds: combatClamp(
          combatFinite(options.blendSeconds, 0.14),
          0,
          0.8
        )
      });
      return false;
    }

    const aliases = ANIMATION_NAME_MAP[semanticState] ??
      [String(semanticState).toLowerCase()];

    const selected = groups.find(group => {
      const name = String(group.name ?? "").toLowerCase();
      return aliases.some(alias => name.includes(alias));
    });

    if (!selected) {
      runtime.events.emit("combat:animationMissing", {
        combatantId: record.id,
        state: semanticState,
        availableGroups: groups.map(group => group.name)
      });
      return false;
    }

    const previous = record.activeAnimationGroup;
    const loop = LOOPING_ANIMATION_STATES.has(semanticState);

    try {
      if (selected !== previous && !selected.isPlaying) {
        selected.start(loop, 1.0);
      } else if (selected.isPlaying && !loop && options.restart === true) {
        selected.restart();
      }

      selected.setWeightForAllAnimatables?.(
        previous && previous !== selected ? 0 : 1
      );

      record.animationBlendState = {
        from: previous && previous !== selected ? previous : null,
        to: selected,
        elapsed: 0,
        duration: combatClamp(
          combatFinite(options.blendSeconds, 0.14),
          0.02,
          0.8
        )
      };

      if (!previous || previous === selected) {
        selected.setWeightForAllAnimatables?.(1);
        record.activeAnimationGroup = selected;
        record.animationBlendState = null;
      }

      runtime.events.emit("combat:animationStateChanged", {
        combatantId: record.id,
        state: semanticState,
        animationGroup: selected.name,
        blendedFrom: previous?.name ?? null
      });

      return true;
    } catch (error) {
      logger.warn("Animation transition failed.", error);
      return false;
    }
  }

  function registerAnimations(id, animationGroups) {
    const record = getCombatant(id);

    if (!record || !Array.isArray(animationGroups)) {
      return { ok: false, reason: "invalid-combatant-or-groups" };
    }

    for (const group of record.animationGroups) {
      try {
        if (group.isPlaying) group.stop();
      } catch {}
    }

    record.animationGroups = animationGroups.filter(
      group => group && typeof group.start === "function"
    );
    record.activeAnimationGroup = null;
    record.animationBlendState = null;

    return {
      ok: true,
      groups: record.animationGroups.length
    };
  }

  function findCover(id, maxDistance = 3.2) {
    const record = getCombatant(id);
    const position = record && entityPosition(record);

    if (!record || !position) {
      return { ok: false, reason: "invalid-character-position" };
    }

    const forward = entityForward(record);
    const candidates = [
      forward,
      new B.Vector3(forward.z, 0, -forward.x),
      new B.Vector3(-forward.z, 0, forward.x),
      forward.scale(-1)
    ];

    let best = null;

    for (const direction of candidates) {
      const ray = new B.Ray(
        position.add(new B.Vector3(0, 1.1, 0)),
        direction.normalize(),
        maxDistance
      );

      const hit = scene.pickWithRay(
        ray,
        mesh =>
          mesh.isEnabled() &&
          mesh.isVisible &&
          mesh.isPickable !== false &&
          (mesh.metadata?.isCover === true ||
           mesh.metadata?.blocksPlayer === true)
      );

      if (!hit?.hit || !hit.pickedPoint) continue;

      const normal = hit.getNormal(true, true);
      if (!normal) continue;

      // A wall-like surface must have a near-horizontal normal.
      if (Math.abs(normal.y) > 0.42) continue;

      const candidate = {
        id: hit.pickedMesh?.metadata?.coverId ??
          hit.pickedMesh?.uniqueId ??
          `cover-${hit.pickedPoint.x}-${hit.pickedPoint.z}`,
        mesh: hit.pickedMesh,
        point: hit.pickedPoint.clone(),
        normal: normal.clone(),
        distanceM: hit.distance
      };

      if (!best || candidate.distanceM < best.distanceM) {
        best = candidate;
      }
    }

    if (!best) {
      return { ok: false, reason: "no-cover-surface" };
    }

    return { ok: true, cover: best };
  }

  function enterCover(id, requestedCover = null) {
    const record = getCombatant(id);
    if (!record || record.dead || record.incapacitated) {
      return { ok: false, reason: "character-unavailable" };
    }

    const result = requestedCover
      ? { ok: true, cover: requestedCover }
      : findCover(id);

    if (!result.ok) return result;

    const cover = result.cover;

    coverStates.set(id, {
      coverId: cover.id,
      point: cover.point.clone(),
      normal: cover.normal.clone(),
      mesh: cover.mesh ?? null,
      enteredAt: elapsed
    });

    // Place the character immediately outside the wall and face into it.
    const safeNormal = cover.normal.clone();
    safeNormal.y = 0;

    if (safeNormal.lengthSquared() > 1e-5) {
      safeNormal.normalize();
      const snapped = cover.point.add(safeNormal.scale(0.46));
      const position = entityPosition(record);
      snapped.y = position?.y ?? snapped.y;

      if (record.root) record.root.position.copyFrom(snapped);

      if (record.position?.copyFrom) {
        record.position.copyFrom(snapped);
      } else if (record.position) {
        Object.assign(record.position, {
          x: snapped.x,
          y: snapped.y,
          z: snapped.z
        });
      }

      record.heading = Math.atan2(-safeNormal.x, -safeNormal.z);
      if (record.root) record.root.rotation.y = record.heading;
    }

    record.coverId = cover.id;
    record.state = "inCover";

    runtime.events.emit("combat:coverEntered", {
      characterId: id,
      coverId: cover.id,
      position: cover.point.clone()
    });

    playAnimation(record, "cover");

    return { ok: true, coverId: cover.id };
  }

  function moveAlongCover(id, amountM) {
    const record = getCombatant(id);
    const cover = coverStates.get(id);

    if (!record || !cover || !entityPosition(record)) {
      return { ok: false, reason: "not-in-cover" };
    }

    const direction = new B.Vector3(
      cover.normal.z,
      0,
      -cover.normal.x
    );

    const position = entityPosition(record).add(
      direction.scale(combatClamp(amountM, -1.2, 1.2))
    );

    const ray = new B.Ray(
      position.add(new B.Vector3(0, 1, 0)),
      cover.normal.scale(-1),
      1.3
    );

    const hit = scene.pickWithRay(
      ray,
      mesh => mesh === cover.mesh
    );

    if (!hit?.hit) {
      releaseFromCover(id);
      return { ok: false, reason: "cover-surface-ended" };
    }

    if (record.root) record.root.position.copyFrom(position);

    if (record.position?.copyFrom) {
      record.position.copyFrom(position);
    } else if (record.position) {
      Object.assign(record.position, {
        x: position.x,
        y: position.y,
        z: position.z
      });
    }

    return { ok: true, position };
  }

  function releaseFromCover(id) {
    const cover = coverStates.get(id);
    const record = getCombatant(id);

    if (!cover) return false;

    coverStates.delete(id);

    if (record) {
      record.coverId = null;

      if (!record.dead && !record.incapacitated) {
        record.state = "neutral";
      }
    }

    runtime.events.emit("combat:coverExited", {
      characterId: id,
      coverId: cover.coverId
    });

    return true;
  }

  function setCoverAim(id, aiming) {
    const record = getCombatant(id);
    const cover = coverStates.get(id);

    if (!record || !cover) {
      return { ok: false, reason: "not-in-cover" };
    }

    record.metadata.aimingFromCover = aiming === true;

    if (aiming) playAnimation(record, "aim");
    else playAnimation(record, "cover");

    runtime.events.emit("combat:coverAimChanged", {
      characterId: id,
      coverId: cover.coverId,
      aiming: record.metadata.aimingFromCover
    });

    return { ok: true, aiming: record.metadata.aimingFromCover };
  }

  function transitionCover(id, direction = 1) {
    const current = coverStates.get(id);
    if (!current) return { ok: false, reason: "not-in-cover" };

    const record = getCombatant(id);
    if (!record) return { ok: false, reason: "unknown-combatant" };

    const tangent = new B.Vector3(current.normal.z, 0, -current.normal.x)
      .scale(Math.sign(combatFinite(direction, 1)) || 1);

    const origin = entityPosition(record);
    if (!origin) return { ok: false, reason: "invalid-character-position" };

    const ray = new B.Ray(
      origin.add(new B.Vector3(0, 1.0, 0)),
      tangent,
      1.6
    );

    const hit = scene.pickWithRay(
      ray,
      mesh =>
        mesh?.isEnabled() &&
        mesh?.isVisible &&
        mesh?.isPickable !== false &&
        (mesh.metadata?.isCover === true ||
         mesh.metadata?.blocksPlayer === true)
    );

    if (!hit?.hit || !hit.pickedPoint) {
      releaseFromCover(id);
      return { ok: false, reason: "cover-transition-unavailable" };
    }

    const normal = hit.getNormal(true, true);
    if (!normal || Math.abs(normal.y) > 0.42) {
      return { ok: false, reason: "invalid-adjacent-cover" };
    }

    const next = {
      id: hit.pickedMesh?.metadata?.coverId ??
        hit.pickedMesh?.uniqueId ??
        `cover-${hit.pickedPoint.x}-${hit.pickedPoint.z}`,
      mesh: hit.pickedMesh,
      point: hit.pickedPoint.clone(),
      normal: normal.clone(),
      distanceM: hit.distance
    };

    releaseFromCover(id);
    return enterCover(id, next);
  }

  function setHostage(holderId, targetId) {
    const holder = getCombatant(holderId);
    const target = getCombatant(targetId);

    if (!holder || !target || holderId === targetId) {
      return { ok: false, reason: "invalid-participants" };
    }

    if (holder.dead || holder.incapacitated || target.dead) {
      return { ok: false, reason: "participant-unavailable" };
    }

    if (
      target.state !== "stunned" &&
      target.metadata.takedownVulnerable !== true
    ) {
      return { ok: false, reason: "target-not-subdued" };
    }

    const a = entityPosition(holder);
    const b = entityPosition(target);

    if (!a || !b || B.Vector3.Distance(a, b) > 1.45) {
      return { ok: false, reason: "out-of-range" };
    }

    if (holder.hostageTargetId || target.hostageTargetId || target.carriedBy) {
      return { ok: false, reason: "participant-already-engaged" };
    }

    holder.hostageTargetId = targetId;
    holder.state = "holdingHostage";
    target.state = "hostage";
    target.incapacitated = true;
    target.metadata.hostageHolderId = holderId;

    runtime.events.emit("combat:hostageStarted", {
      holderId,
      targetId
    });

    return { ok: true };
  }

  function releaseHostage(holderId, options = {}) {
    const holder = getCombatant(holderId);
    if (!holder?.hostageTargetId) {
      return { ok: false, reason: "no-hostage" };
    }

    const target = getCombatant(holder.hostageTargetId);
    const targetId = holder.hostageTargetId;

    holder.hostageTargetId = null;
    holder.state = "neutral";

    if (target) {
      target.incapacitated = options.subdued === true;
      target.state = target.dead
        ? "dead"
        : (options.subdued ? "stunned" : "neutral");
      delete target.metadata.hostageHolderId;
    }

    runtime.events.emit("combat:hostageReleased", {
      holderId,
      targetId,
      subdued: options.subdued === true
    });

    return { ok: true, targetId };
  }

  function carryBody(carrierId, targetId) {
    const carrier = getCombatant(carrierId);
    const target = getCombatant(targetId);

    if (!carrier || !target || carrierId === targetId) {
      return { ok: false, reason: "invalid-participants" };
    }

    if (!target.dead && !target.incapacitated) {
      return { ok: false, reason: "target-not-incapacitated" };
    }

    if (target.carriedBy) {
      return { ok: false, reason: "already-carried" };
    }

    if (
      carrier.metadata.carryingBody ||
      carrier.dead ||
      carrier.incapacitated ||
      carrier.hostageTargetId
    ) {
      return { ok: false, reason: "carrier-unavailable" };
    }

    const targetMass = combatClamp(combatFinite(target.bodyMassKg, 75), 1, 500);
    const maxCarryMass = combatClamp(
      combatFinite(carrier.metadata.maxCarryMassKg, 125),
      20,
      250
    );

    if (targetMass > maxCarryMass) {
      return { ok: false, reason: "carry-weight-limit" };
    }

    const a = entityPosition(carrier);
    const b = entityPosition(target);

    if (!a || !b || B.Vector3.Distance(a, b) > 1.8) {
      return { ok: false, reason: "out-of-range" };
    }

    const anchor = new B.TransformNode(
      `body-carry-anchor-${carrierId}-${targetId}`,
      scene
    );

    anchor.parent = carrier.root ?? null;
    anchor.position.set(0.15, 0.45, -0.62);

    if (target.root) {
      target.root.parent = anchor;
      target.root.position.set(0, 0, 0);
      target.root.rotation.set(Math.PI / 2, 0, 0);
    } else {
      anchor.dispose();
      return { ok: false, reason: "target-root-unavailable" };
    }

    target.carriedBy = carrierId;
    carrier.metadata.carryingBody = targetId;
    carrier.state = "carryingBody";
    target.state = "beingCarried";

    carriedBodies.set(targetId, {
      carrierId,
      targetId,
      anchor,
      previousParent: null,
      lastWorldPosition: b.clone(),
      bodyMassKg: targetMass
    });

    runtime.events.emit("combat:bodyPickedUp", {
      carrierId,
      targetId
    });

    playAnimation(carrier, "carry");

    return { ok: true };
  }

  function releaseCarriedBody(carrierOrTargetId, options = {}) {
    let targetId = carriedBodies.has(carrierOrTargetId)
      ? carrierOrTargetId
      : null;

    if (!targetId) {
      targetId = [...carriedBodies.values()].find(
        record => record.carrierId === carrierOrTargetId
      )?.targetId ?? null;
    }

    if (!targetId) return false;

    const carrying = carriedBodies.get(targetId);
    const carrier = getCombatant(carrying.carrierId);
    const target = getCombatant(targetId);

    const position = carrying.anchor.getAbsolutePosition().clone();

    if (target?.root) {
      target.root.parent = null;
      target.root.position.copyFrom(position);
      target.root.rotation.set(0, 0, 0);

      if (options.hidden === true) {
        target.metadata.hidden = true;
        target.root.setEnabled(false);
      } else {
        target.metadata.hidden = false;
        target.root.setEnabled(true);
      }
    }

    if (target?.position?.copyFrom) {
      target.position.copyFrom(position);
    } else if (target?.position) {
      Object.assign(target.position, {
        x: position.x,
        y: position.y,
        z: position.z
      });
    }

    carrying.anchor.dispose();
    carriedBodies.delete(targetId);

    if (carrier) {
      delete carrier.metadata.carryingBody;

      if (!carrier.dead && !carrier.incapacitated) {
        carrier.state = "neutral";
      }
    }

    if (target) {
      target.carriedBy = null;
      target.state = target.dead ? "dead" : "incapacitated";
    }

    runtime.events.emit("combat:bodyDropped", {
      carrierId: carrying.carrierId,
      targetId,
      position,
      hidden: options.hidden === true
    });

    return true;
  }

  function revealBody(targetId) {
    const target = getCombatant(targetId);
    if (!target || !target.metadata.hidden) {
      return { ok: false, reason: "body-not-hidden" };
    }

    target.metadata.hidden = false;
    target.root?.setEnabled(true);

    runtime.events.emit("combat:bodyRevealed", {
      targetId,
      position: entityPosition(target)
    });

    return { ok: true, targetId };
  }

  function recordReplaySample() {
    const player = getPlayerService();
    const id = player?.getActiveCharacterId?.();
    const character = getCombatant(id);

    if (!character) return;

    const position = entityPosition(character);
    if (!position) return;

    replayConfig.samples.push({
      t: elapsed,
      characterId: id,
      position: position.clone(),
      heading: character.heading,
      cameraPosition: camera.position.clone(),
      cameraTarget: camera.getTarget().clone()
    });

    const minimumTime = elapsed - replayConfig.maxSeconds;

    while (
      replayConfig.samples.length &&
      replayConfig.samples[0].t < minimumTime
    ) {
      replayConfig.samples.shift();
    }
  }

  function startReplay(options = {}) {
    if (replayConfig.replay) {
      return { ok: false, reason: "replay-already-running" };
    }

    if (replayConfig.samples.length < 2) {
      return { ok: false, reason: "insufficient-recorded-samples" };
    }

    const end = replayConfig.samples.length - 1;
    const requestedSeconds = combatClamp(
      combatFinite(options.seconds, 8),
      1,
      replayConfig.maxSeconds
    );

    const endTime = replayConfig.samples[end].t;
    const startTime = Math.max(
      replayConfig.samples[0].t,
      endTime - requestedSeconds
    );

    replayConfig.replay = {
      startTime,
      endTime,
      cursor: startTime,
      speed: combatClamp(combatFinite(options.speed, 0.4), 0.05, 1),
      savedCameraPosition: camera.position.clone(),
      savedCameraTarget: camera.getTarget().clone(),
      finished: false
    };

    runtime.events.emit("replay:started", {
      startTime,
      endTime,
      speed: replayConfig.replay.speed
    });

    return { ok: true, startTime, endTime };
  }

  function interpolateReplaySample(time) {
    const samples = replayConfig.samples;

    if (!samples.length) return null;

    let upper = samples.findIndex(sample => sample.t >= time);

    if (upper < 0) upper = samples.length - 1;
    if (upper === 0) return samples[0];

    const a = samples[upper - 1];
    const b = samples[upper];

    const t = combatClamp(
      (time - a.t) / Math.max(0.0001, b.t - a.t),
      0,
      1
    );

    let headingDelta = b.heading - a.heading;
    while (headingDelta > Math.PI) headingDelta -= Math.PI * 2;
    while (headingDelta < -Math.PI) headingDelta += Math.PI * 2;

    return {
      t: time,
      characterId: b.characterId,
      position: B.Vector3.Lerp(a.position, b.position, t),
      heading: a.heading + headingDelta * t,
      cameraPosition: B.Vector3.Lerp(
        a.cameraPosition,
        b.cameraPosition,
        t
      ),
      cameraTarget: B.Vector3.Lerp(
        a.cameraTarget,
        b.cameraTarget,
        t
      )
    };
  }

  function stopReplay() {
    const replay = replayConfig.replay;
    if (!replay) return false;

    replayConfig.replay = null;
    const player = getPlayerService();
    const character = player?.getCharacter?.(player?.getActiveCharacterId?.());

    if (character?.position) {
      const heading = combatFinite(character.heading, 0);
      const targetPosition = new B.Vector3(
        character.position.x,
        character.position.y + 1.15,
        character.position.z
      );

      camera.position.copyFrom(new B.Vector3(
        character.position.x - Math.sin(heading) * 5.5,
        character.position.y + 2.8,
        character.position.z - Math.cos(heading) * 5.5
      ));
      camera.setTarget(targetPosition);
    } else {
      camera.position.copyFrom(replay.savedCameraPosition);
      camera.setTarget(replay.savedCameraTarget);
    }

    runtime.events.emit("replay:stopped", {});
    return true;
  }

  function updateReplay(dt) {
    const replay = replayConfig.replay;
    if (!replay) return;

    replay.cursor += dt * replay.speed;

    const sample = interpolateReplaySample(replay.cursor);

    if (sample) {
      const offset = new B.Vector3(
        Math.sin(sample.heading + 0.55) * 4.5,
        2.2,
        Math.cos(sample.heading + 0.55) * 4.5
      );

      camera.position.copyFrom(sample.position.add(offset));
      camera.setTarget(
        sample.position.add(new B.Vector3(0, 1.0, 0))
      );
    }

    if (replay.cursor >= replay.endTime) {
      stopReplay();
    }
  }

  /* --------------------------------------------------------------- */
  /* Lifecycle                                                         */
  /* --------------------------------------------------------------- */

  async function initialize() {
    if (initialized) return { ok: true };

    initialized = true;

    registerDefaultPlayerCharacters();

    serviceDisposers.push(runtime.services.register("combat", {
      registerCombatant,
      unregisterCombatant,
      getCombatant,

      beginAttack,
      cancelAttack,
      applyDamage,

      enableRagdoll,
      recoverFromRagdoll,
      setDefenseState,
      revealBody,

      playAnimation,
      registerAnimations,

      findCover,
      enterCover,
      moveAlongCover,
      transitionCover,
      setCoverAim,
      releaseFromCover,

      setHostage,
      releaseHostage,
      carryBody,
      releaseCarriedBody,

      startReplay,
      stopReplay,

      getSnapshot: () => ({
        combatants: combatants.size,
        activeAttacks: activeAttacks.size,
        ragdolls: ragdollHandles.size,
        bodiesCarried: carriedBodies.size,
        replaySamples: replayConfig.samples.length,
        replayEvents: replayConfig.events.length,
        replayActive: Boolean(replayConfig.replay),
        activeDefenses: [...combatants.values()].filter(record =>
          record.defense.blocking || elapsed < record.defense.dodgingUntil
        ).length
      })
    }));

    eventDisposers.push(
      runtime.events.on("player:characterChanged", () => {
        registerDefaultPlayerCharacters();
      })
    );

    eventDisposers.push(
      runtime.events.on("player:respawned", payload => {
        const activeId = payload?.characterId ??
          getPlayerService()?.getActiveCharacterId?.();
        const record = getCombatant(activeId);
        if (!record) return;

        const profile = getPlayerService()?.getCharacter?.(record.id);
        if (profile) {
          record.health = combatClamp(
            combatFinite(profile.health, record.maxHealth),
            1,
            record.maxHealth
          );
          record.stamina = combatClamp(
            combatFinite(profile.stamina, 100),
            0,
            100
          );
        } else {
          record.health = record.maxHealth;
        }

        record.dead = false;
        record.incapacitated = false;
        record.state = "neutral";
        record.ragdollEnabled = false;
        record.root?.setEnabled(true);

        try {
          ragdollHandles.get(record.id)?.dispose?.();
        } catch {}

        ragdollHandles.delete(record.id);
      })
    );

    eventDisposers.push(
      runtime.events.on("combat:hit", payload => {
        const target = getCombatant(payload.targetId);
        if (target) playAnimation(target, "hitReaction");
      })
    );

    eventDisposers.push(
      runtime.events.on("combat:death", payload => {
        const target = getCombatant(payload.targetId);
        if (target) playAnimation(target, "dead");
      })
    );

    for (const eventName of [
      "combat:hit", "combat:death", "weapon:fired", "vehicle:damaged",
      "mission:started", "mission:completed", "mission:failed", "police:wantedChanged"
    ]) {
      eventDisposers.push(
        runtime.events.on(eventName, payload => {
          recordReplayEvent(eventName, payload ?? {});
        })
      );
    }

    logger.info("Combat and animation system initialized.");

    return { ok: true };
  }

  function start() {
    running = true;
    return { ok: true };
  }

  function update(context) {
    if (!initialized || !running || disposed) return;

    const dt = combatClamp(
      combatFinite(context.simulationDeltaSeconds ?? context.gameDeltaSeconds),
      0,
      0.1
    );

    elapsed += dt;

    for (const record of combatants.values()) {
      syncPlayerCombatState(record);
      updateAnimationBlend(record, dt);

      if (
        record.defense.blocking &&
        record.state !== "inCover" &&
        record.state !== "dead"
      ) {
        record.state = "blocking";
      }

      if (elapsed >= record.defense.dodgingUntil) {
        record.defense.dodgingUntil = -Infinity;
      }
    }

    for (const [attackerId, attack] of activeAttacks) {
      const attacker = getCombatant(attackerId);

      if (!attacker || attacker.dead || attacker.incapacitated) {
        cancelAttack(attackerId, "attacker-state-invalid");
        continue;
      }

      attack.phaseElapsed += dt;

      if (
        attack.phase === "windup" &&
        attack.phaseElapsed >= attack.config.windup
      ) {
        attack.phase = "active";
        attack.phaseElapsed = 0;

        runtime.events.emit("combat:attackActive", {
          attackId: attack.id,
          attackerId
        });
      }

      if (
        attack.phase === "active" &&
        attack.phaseElapsed <= attack.config.active
      ) {
        performAttackHit(attack);
      } else if (
        attack.phase === "active" &&
        attack.phaseElapsed > attack.config.active
      ) {
        attack.phase = "recovery";
        attack.phaseElapsed = 0;
      }

      if (
        attack.phase === "recovery" &&
        attack.phaseElapsed >= attack.config.recovery
      ) {
        activeAttacks.delete(attackerId);
        attacker.cooldownUntil = elapsed + 0.12;

        if (!attacker.dead && !attacker.incapacitated) {
          attacker.state = "neutral";
        }

        runtime.events.emit("combat:attackCompleted", {
          attackId: attack.id,
          attackerId,
          targetsHit: [...attack.hitTargets]
        });
      }
    }

    replayConfig.sampleAccumulator += dt;

    if (
      replayConfig.recording &&
      replayConfig.sampleAccumulator >= replayConfig.sampleInterval
    ) {
      replayConfig.sampleAccumulator %= replayConfig.sampleInterval;
      recordReplaySample();
    }

    updateReplay(dt);

    for (const [targetId, carried] of carriedBodies) {
      if (!carried.anchor || carried.anchor.isDisposed()) {
        carriedBodies.delete(targetId);
      }
    }
  }

  function stop() {
    running = false;

    for (const id of activeAttacks.keys()) {
      cancelAttack(id, "system-stopped");
    }

    stopReplay();
    return { ok: true };
  }

  function dispose() {
    if (disposed) return;

    disposed = true;
    running = false;

    stopReplay();

    for (const off of eventDisposers.splice(0)) {
      try { off(); } catch {}
    }

    for (const off of serviceDisposers.splice(0)) {
      try { off(); } catch {}
    }

    for (const handle of ragdollHandles.values()) {
      try { handle.dispose?.(); } catch {}
    }

    for (const record of combatants.values()) {
      disposeHitboxProxies(record);
    }

    try {
      hitProxyMaterial.dispose();
    } catch {}

    for (const carried of carriedBodies.values()) {
      try { carried.anchor.dispose(); } catch {}
    }

    activeAttacks.clear();
    ragdollHandles.clear();
    carriedBodies.clear();
    coverStates.clear();
    combatants.clear();
    replayConfig.samples.length = 0;
    replayConfig.events.length = 0;
  }

  return Object.freeze({
    initialize,
    start,
    update,
    stop,
    dispose
  });
}
