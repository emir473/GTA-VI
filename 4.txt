function finite(value, fallback = 0) {
  return Number.isFinite(value) ? value : fallback;
}

function clamp(value, min, max) {
  return Math.max(min, Math.min(max, finite(value)));
}

function lerp(a, b, t) {
  return a + (b - a) * t;
}

function smoothstep01(t) {
  const v = clamp(t, 0, 1);
  return v * v * (3 - 2 * v);
}

function normalizeXZ(B, x, z) {
  const length = Math.hypot(x, z) || 1;
  return new B.Vector3(x / length, 0, z / length);
}

function hash32(value) {
  let x = value | 0;
  x = Math.imul(x ^ (x >>> 16), 0x7feb352d);
  x = Math.imul(x ^ (x >>> 15), 0x846ca68b);
  return (x ^ (x >>> 16)) >>> 0;
}

function seededRandom(seed) {
  let value = seed >>> 0;

  return () => {
    value = (value + 0x6D2B79F5) >>> 0;

    let t = value;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);

    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function smoothNoise2D(x, z, seed = 7117) {
  const ix = Math.floor(x);
  const iz = Math.floor(z);

  const fx = x - ix;
  const fz = z - iz;

  const sx = smoothstep01(fx);
  const sz = smoothstep01(fz);

  const sample = (a, b) =>
    hash32(Math.imul(a, 374761393) ^
           Math.imul(b, 668265263) ^
           seed) / 4294967295;

  const a = sample(ix, iz);
  const b = sample(ix + 1, iz);
  const c = sample(ix, iz + 1);
  const d = sample(ix + 1, iz + 1);

  return lerp(lerp(a, b, sx), lerp(c, d, sx), sz);
}

function interpolateWeather(from, to, t) {
  const result = {};

  for (const key of Object.keys(WEATHER_PROFILES.sunny)) {
    result[key] = lerp(from[key], to[key], t);
  }

  return result;
}

function dayNumberUTC(date = new Date()) {
  const start = Date.UTC(date.getUTCFullYear(), 0, 0);
  const now = Date.UTC(
    date.getUTCFullYear(),
    date.getUTCMonth(),
    date.getUTCDate()
  );

  return Math.floor((now - start) / 86400000);
}

function calculateSolarDirection(B, timeMinutes, latitudeRad, dayOfYear) {
  const dayAngle = (2 * Math.PI * (dayOfYear - 81)) / 365.2422;

  // A compact seasonal declination approximation, not an ephemeris.
  const declination = 0.4091 * Math.sin(dayAngle);

  const localHour = ((timeMinutes / 60) % 24 + 24) % 24;
  const hourAngle = ((localHour - 12) * Math.PI) / 12;

  const sinAltitude =
    Math.sin(latitudeRad) * Math.sin(declination) +
    Math.cos(latitudeRad) * Math.cos(declination) *
      Math.cos(hourAngle);

  const altitude = Math.asin(clamp(sinAltitude, -1, 1));

  const x = Math.cos(declination) * Math.sin(hourAngle);

  const y = sinAltitude;

  const z =
    Math.sin(declination) * Math.cos(latitudeRad) -
    Math.cos(declination) * Math.cos(hourAngle) *
      Math.sin(latitudeRad);

  const direction = new B.Vector3(x, y, z).normalize();

  return {
    direction,
    altitude,
    declination,
    hourAngle,
    localHour
  };
}

function makeCloudTexture(B, scene, seed, name) {
  const size = 512;
  const canvas = document.createElement("canvas");
  canvas.width = size;
  canvas.height = size;

  const context = canvas.getContext("2d");
  if (!context) return null;

  const random = seededRandom(seed);

  context.clearRect(0, 0, size, size);

  // Overlapping blurred density lobes form a procedural cloud texture.
  for (let i = 0; i < 135; i++) {
    const x = random() * size;
    const y = random() * size;
    const radius = 10 + random() * 72;

    const gradient = context.createRadialGradient(
      x, y, radius * 0.05,
      x, y, radius
    );

    const alpha = 0.035 + random() * 0.12;

    gradient.addColorStop(0, `rgba(245,248,255,${alpha})`);
    gradient.addColorStop(0.48, `rgba(218,231,244,${alpha * 0.65})`);
    gradient.addColorStop(1, "rgba(205,220,238,0)");

    context.fillStyle = gradient;
    context.fillRect(x - radius, y - radius, radius * 2, radius * 2);
  }

  let texture;

  try {
    texture = new B.DynamicTexture(
      name,
      canvas,
      scene,
      false
    );
  } catch {
    return null;
  }

  texture.hasAlpha = true;
  texture.wrapU = B.Texture.WRAP_ADDRESSMODE;
  texture.wrapV = B.Texture.WRAP_ADDRESSMODE;
  texture.update(false);

  return texture;
}

function createRainTexture(B, scene) {
  const canvas = document.createElement("canvas");

  canvas.width = 4;
  canvas.height = 32;

  const context = canvas.getContext("2d");
  if (!context) return null;

  context.clearRect(0, 0, 4, 32);
  context.strokeStyle = "rgba(218,238,255,0.85)";
  context.lineWidth = 1.2;
  context.beginPath();
  context.moveTo(2, 1);
  context.lineTo(2, 30);
  context.stroke();

  try {
    const texture = new B.DynamicTexture(
      "leonida-rain-texture",
      canvas,
      scene,
      false
    );

    texture.hasAlpha = true;
    texture.update(false);
    return texture;
  } catch {
    // Rain is an optional visual effect; failure to create its texture must
    // not prevent the core world and weather state machine from starting.
    return null;
  }
}

function createSkyGradientMesh(B, scene) {
  const dome = B.MeshBuilder.CreateSphere(
    "leonida-sky-dome",
    {
      diameter: 9000,
      segments: 32
    },
    scene
  );

  dome.isPickable = false;
  dome.checkCollisions = false;

  const material = new B.StandardMaterial("leonida-sky-material", scene);
  material.backFaceCulling = false;
  material.disableLighting = true;
  material.useVertexColors = true;
  material.emissiveColor = B.Color3.White();
  material.specularColor = B.Color3.Black();

  dome.material = material;

  const positions = dome.getVerticesData(B.VertexBuffer.PositionKind) || [];
  const colors = [];

  for (let i = 0; i < positions.length; i += 3) {
    const y = positions[i + 1];
    const normalized = clamp((y + 4500) / 9000, 0, 1);

    // Placeholder colors are overwritten by updateSky().
    colors.push(normalized, normalized, normalized, 1);
  }

  dome.setVerticesData(B.VertexBuffer.ColorKind, colors, false);

  return {
    dome,
    material,

    updateSky(palette) {
      const colorData = [];

      const bottom = B.Color3.FromHexString(palette.horizon);
      const mid = B.Color3.FromHexString(palette.mid);
      const top = B.Color3.FromHexString(palette.zenith);

      for (let i = 0; i < positions.length; i += 3) {
        const y = positions[i + 1];
        const n = clamp((y + 4500) / 9000, 0, 1);

        let color;

        if (n < 0.46) {
          color = B.Color3.Lerp(bottom, mid, n / 0.46);
        } else {
          color = B.Color3.Lerp(
            mid,
            top,
            (n - 0.46) / 0.54
          );
        }

        colorData.push(color.r, color.g, color.b, 1);
      }

      dome.updateVerticesData(B.VertexBuffer.ColorKind, colorData);
    }
  };
}

function createStarField(B, scene) {
  const master = B.MeshBuilder.CreateSphere(
    "leonida-star-master",
    { diameter: 1.3, segments: 4 },
    scene
  );

  master.isPickable = false;
  master.alwaysSelectAsActiveMesh = true;

  const material = new B.StandardMaterial("leonida-star-material", scene);
  material.disableLighting = true;
  material.emissiveColor = new B.Color3(0.74, 0.82, 1);
  material.specularColor = B.Color3.Black();

  master.material = material;

  const random = seededRandom(44125);
  const transforms = new Float32Array(450 * 16);
  const radius = 4100;

  for (let index = 0; index < 450; index++) {
    const theta = random() * Math.PI * 2;
    const y = 2 * random() - 1;
    const horizontal = Math.sqrt(Math.max(0, 1 - y * y));

    const position = new B.Vector3(
      radius * horizontal * Math.cos(theta),
      radius * y,
      radius * horizontal * Math.sin(theta)
    );

    const scale = 0.65 + random() * 1.5;

    const matrix = B.Matrix.Compose(
      new B.Vector3(scale, scale, scale),
      B.Quaternion.Identity(),
      position
    );

    matrix.copyToArray(transforms, index * 16);
  }

  master.thinInstanceSetBuffer("matrix", transforms, 16);
  master.thinInstanceCount = 450;

  return {
    mesh: master,

    update(center, visibility) {
      master.position.copyFrom(center);
      master.visibility = clamp(visibility, 0, 1);
      master.setEnabled(visibility > 0.01);
    }
  };
}

function createCloudLayer(B, scene, seed, altitude, scaleX, scaleZ) {
  const mesh = B.MeshBuilder.CreateGround(
    `leonida-cloud-layer-${seed}`,
    {
      width: scaleX,
      height: scaleZ,
      subdivisions: 1
    },
    scene
  );

  mesh.rotation.x = Math.PI;
  mesh.position.y = altitude;
  mesh.isPickable = false;
  mesh.checkCollisions = false;

  const texture = makeCloudTexture(
    B,
    scene,
    seed,
    `leonida-cloud-texture-${seed}`
  );

  const material = new B.StandardMaterial(
    `leonida-cloud-material-${seed}`,
    scene
  );

  material.backFaceCulling = false;
  material.disableLighting = true;
  material.emissiveColor = new B.Color3(0.85, 0.89, 0.96);
  material.diffuseColor = new B.Color3(0.9, 0.93, 0.97);
  material.alpha = 0.25;

  if (texture) {
    material.diffuseTexture = texture;
    material.opacityTexture = texture;
  }

  mesh.material = material;

  return {
    mesh,
    material,
    texture,
    baseOpacity: material.alpha,

    setCoverage(coverage) {
      const c = clamp(coverage, 0, 1);
      material.alpha = 0.07 + 0.60 * c;
    },

    update(dt, windSpeed, cameraPosition) {
      mesh.position.x = cameraPosition.x;
      mesh.position.z = cameraPosition.z;

      if (texture) {
        texture.uOffset =
          (texture.uOffset + dt * windSpeed * 0.0008) % 1;
      }
    },

    dispose() {
      mesh.dispose();
      material.dispose();
      texture?.dispose();
    }
  };
}

function createAdaptiveCloudLayer(B, scene, graphics, seed, altitude, scaleX, scaleZ) {
  if (typeof graphics?.createCloudLayer === "function") {
    try {
      const cloud = graphics.createCloudLayer({
        name: `leonida-weather-cloud-${seed}`,
        seed,
        altitudeMin: altitude,
        altitudeMax: altitude + 95,
        width: scaleX,
        depth: scaleZ,
        opacity: 0.24,
        density: 0.45
      });

      if (cloud && typeof cloud.update === "function" &&
          typeof cloud.setDensity === "function" &&
          typeof cloud.setWind === "function") {
        let lastWind = { x: 0.8, z: 0.15 };
        return Object.freeze({
          setCoverage(coverage) {
            const value = clamp(coverage, 0, 1);
            cloud.setDensity(Math.max(0.03, value));
            cloud.setOpacity?.(clamp(0.07 + value * 0.68, 0.07, 0.82));
          },
          setOpacity(opacity) {
            cloud.setOpacity?.(clamp(finite(opacity, 0.24), 0.05, 0.95));
          },
          update(dt, windSpeed, _cameraPosition, directionRad = 0.68) {
            lastWind = {
              x: Math.cos(finite(directionRad, 0.68)) * Math.max(0, finite(windSpeed)),
              z: Math.sin(finite(directionRad, 0.68)) * Math.max(0, finite(windSpeed))
            };
            cloud.setWind(lastWind);
            cloud.update(dt);
          },
          dispose() {
            cloud.dispose();
          }
        });
      }

      cloud?.dispose?.();
      throw new Error("Graphics cloud factory returned an incompatible cloud layer.");
    } catch (error) {
      console.warn("Shared 3D cloud layer unavailable; using weather fallback.", error);
    }
  }

  return createCloudLayer(B, scene, seed, altitude, scaleX, scaleZ);
}

export async function createWorldWeatherSystem({
  runtime,
  scene,
  camera
}) {
  if (!runtime || !scene || !camera) {
    throw new TypeError(
      "Weather system requires the shared runtime, scene and camera."
    );
  }

  const B = globalThis.BABYLON;

  if (!B) {
    throw new Error("Babylon.js is unavailable for weather simulation.");
  }

  const state = runtime.state;
  const logger = runtime.logger;
  const events = runtime.events;

  let initialized = false;
  let running = false;
  let disposed = false;

  const disposers = [];
  const serviceDisposers = [];

  const tropicalLatitudeRad = 26.1 * Math.PI / 180;

  let seasonDay = dayNumberUTC();
  let timeMinutes = finite(state.get("timeOfDayMinutes"), 9 * 60);

  let currentWeatherId = "sunny";
  let fromParameters = { ...WEATHER_PROFILES.sunny };
  let currentParameters = { ...WEATHER_PROFILES.sunny };
  let targetParameters = { ...WEATHER_PROFILES.sunny };

  let transitionElapsed = 0;
  let transitionDuration = 0;
  let weatherTransitioning = false;

  let globalWetness = 0;
  let elapsedWeatherTime = 0;
  let lastWeatherEventSecond = -1;
  let lastParameterEventTime = -Infinity;
  let lastPublishedMinute = Math.floor(timeMinutes);
  let lastWindSampleTime = -Infinity;

  const weatherZones = new Map();
  let nextWeatherZoneId = 1;

  const solar = {
    direction: new B.Vector3(0.1, 0.9, -0.4),
    altitude: Math.PI / 4,
    localHour: 9,
    declination: 0
  };

  const sceneMaterials = new Set();

  const graphics = runtime.services.get("graphics");
  const sunLight =
    scene.getLightByName("leonida-sun-light");
  const ambientLight =
    scene.getLightByName("leonida-ambient-light");

  const sky = createSkyGradientMesh(B, scene);
  const stars = createStarField(B, scene);

  const sun = B.MeshBuilder.CreateSphere(
    "leonida-visible-sun",
    { diameter: 95, segments: 16 },
    scene
  );

  sun.isPickable = false;
  sun.checkCollisions = false;

  const sunMaterial = new B.StandardMaterial("leonida-sun-material", scene);
  sunMaterial.disableLighting = true;
  sunMaterial.emissiveColor = new B.Color3(1, 0.73, 0.42);
  sunMaterial.diffuseColor = new B.Color3(1, 0.88, 0.68);
  sunMaterial.specularColor = B.Color3.Black();
  sun.material = sunMaterial;

  const moon = B.MeshBuilder.CreateSphere(
    "leonida-visible-moon",
    { diameter: 54, segments: 14 },
    scene
  );

  moon.isPickable = false;
  moon.checkCollisions = false;

  const moonMaterial = new B.StandardMaterial("leonida-moon-material", scene);
  moonMaterial.disableLighting = true;
  moonMaterial.emissiveColor = new B.Color3(0.43, 0.54, 0.72);
  moonMaterial.diffuseColor = new B.Color3(0.48, 0.57, 0.7);
  moonMaterial.specularColor = B.Color3.Black();
  moon.material = moonMaterial;

  const cloudLayers = [
    createAdaptiveCloudLayer(B, scene, graphics, 911, 245, 2600, 2600),
    createAdaptiveCloudLayer(B, scene, graphics, 1289, 310, 3500, 3500)
  ];

  const rainTexture = createRainTexture(B, scene);
  let rainSystem = null;

  if (rainTexture && typeof B.ParticleSystem === "function") {
    rainSystem = new B.ParticleSystem(
      "leonida-rain",
      5000,
      scene
    );

    rainSystem.particleTexture = rainTexture;
    rainSystem.emitter = camera;

    rainSystem.minEmitBox = new B.Vector3(-90, 40, -90);
    rainSystem.maxEmitBox = new B.Vector3(90, 100, 90);

    rainSystem.direction1 = new B.Vector3(-3, -45, 1);
    rainSystem.direction2 = new B.Vector3(3, -65, 5);

    rainSystem.minSize = 0.025;
    rainSystem.maxSize = 0.06;

    rainSystem.minLifeTime = 0.5;
    rainSystem.maxLifeTime = 1.5;

    rainSystem.minEmitPower = 1;
    rainSystem.maxEmitPower = 2;

    rainSystem.updateSpeed = 0.018;
    rainSystem.emitRate = 0;
    rainSystem.gravity = new B.Vector3(0, -9.81, 0);

    rainSystem.color1 = new B.Color4(0.74, 0.84, 0.94, 0.4);
    rainSystem.color2 = new B.Color4(0.82, 0.91, 1, 0.75);
    rainSystem.colorDead = new B.Color4(0.72, 0.82, 0.93, 0);

    rainSystem.minAngularSpeed = 0;
    rainSystem.maxAngularSpeed = 0;

    rainSystem.start();
  }

  function getSunAngles() {
    return {
      altitudeRad: solar.altitude,
      altitudeDeg: solar.altitude * 180 / Math.PI,
      localHour: solar.localHour,
      declinationRad: solar.declination
    };
  }

  function getNightFactor() {
    const fadeRange = 0.12;
    return 1 - smoothstep01(
      clamp((solar.altitude + fadeRange) / (fadeRange * 2), 0, 1)
    );
  }

  function updateSolarModel() {
    const calculation = calculateSolarDirection(
      B,
      timeMinutes,
      tropicalLatitudeRad,
      seasonDay
    );

    solar.direction.copyFrom(calculation.direction);
    solar.altitude = calculation.altitude;
    solar.localHour = calculation.localHour;
    solar.declination = calculation.declination;

    const sunAboveHorizon = solar.altitude > -0.12;
    const nightFactor = getNightFactor();

    sun.position.copyFrom(
      camera.globalPosition.add(solar.direction.scale(3000))
    );

    sun.visibility = smoothstep01(
      clamp((solar.altitude + 0.12) / 0.23, 0, 1)
    );
    sun.setEnabled(sunAboveHorizon);

    const moonDirection = solar.direction.scale(-1);
    moon.position.copyFrom(
      camera.globalPosition.add(moonDirection.scale(2600))
    );
    moon.visibility = nightFactor;
    moon.setEnabled(nightFactor > 0.01);

    if (sunLight) {
      sunLight.direction.copyFrom(solar.direction.scale(-1));

      sunLight.position.copyFrom(
        camera.globalPosition.add(solar.direction.scale(1100))
      );

      const daylight = smoothstep01(
        clamp((solar.altitude + 0.08) / 0.50, 0, 1)
      );

      sunLight.intensity = lerp(0.055, 2.35, daylight) *
        (1 - currentParameters.cloudCover * 0.50);

      const warmth = clamp(
        1 - Math.abs(solar.altitude - 0.13) * 1.9,
        0,
        1
      );

      sunLight.diffuse = B.Color3.Lerp(
        new B.Color3(0.67, 0.76, 0.96),
        new B.Color3(1, 0.72, 0.43),
        warmth
      );
    }

    if (ambientLight) {
      const daylight = smoothstep01(
        clamp((solar.altitude + 0.25) / 0.75, 0, 1)
      );

      ambientLight.intensity = lerp(0.12, 0.8, daylight);
    }

    updateSkyPalette(daylightPalette());
  }

  function daylightPalette() {
    const altitude = solar.altitude;
    const night = getNightFactor();

    const sunrise = 1 - clamp(Math.abs(altitude - 0.12) / 0.36, 0, 1);
    const storm = currentParameters.cloudCover * 0.72;

    function blend(hexA, hexB, amount) {
      return B.Color3.Lerp(
        B.Color3.FromHexString(hexA),
        B.Color3.FromHexString(hexB),
        clamp(amount, 0, 1)
      );
    }

    const horizonDay = blend("#72c8dd", "#f3a078", sunrise);
    const horizon = B.Color3.Lerp(
      horizonDay,
      B.Color3.FromHexString("#28333f"),
      clamp(night * 0.96 + storm * 0.65, 0, 0.95)
    );

    const midDay = blend("#42a9d5", "#b78baf", sunrise * 0.55);
    const mid = B.Color3.Lerp(
      midDay,
      B.Color3.FromHexString("#172231"),
      clamp(night * 0.98 + storm * 0.55, 0, 0.97)
    );

    const topDay = B.Color3.FromHexString("#1476b5");
    const top = B.Color3.Lerp(
      topDay,
      B.Color3.FromHexString("#060e1c"),
      clamp(night * 0.99 + storm * 0.35, 0, 0.99)
    );

    return {
      horizon: horizon.toHexString(),
      mid: mid.toHexString(),
      zenith: top.toHexString()
    };
  }

  function updateSkyPalette(palette) {
    sky.updateSky(palette);
    scene.clearColor = new B.Color4(0.055, 0.08, 0.12, 1);
  }

  function applyFog() {
    const base = currentParameters.fogDensity;
    const nightMultiplier = solar.altitude < 0
      ? 1.35
      : 1;

    scene.fogMode = B.Scene.FOGMODE_EXP2;
    scene.fogDensity = clamp(base * nightMultiplier, 0.000001, 0.01);

    const darkness = getNightFactor();
    const stormFactor = currentParameters.cloudCover;

    const fogDay = B.Color3.FromHexString("#8da4b3");
    const fogNight = B.Color3.FromHexString("#192535");
    const fogStorm = B.Color3.FromHexString("#5e707b");

    const currentFog = B.Color3.Lerp(
      B.Color3.Lerp(fogDay, fogNight, darkness),
      fogStorm,
      stormFactor * 0.65
    );

    scene.fogColor = currentFog;

    graphics?.setAtmosphere?.({
      ambientIntensity: ambientLight?.intensity,
      sunIntensity: sunLight?.intensity,
      fog: {
        mode: "exponential-squared",
        density: scene.fogDensity,
        color: {
          r: currentFog.r,
          g: currentFog.g,
          b: currentFog.b
        }
      }
    });
  }

  function getParameters() {
    return { ...currentParameters };
  }

  function getWeatherProfile(id) {
    return WEATHER_PROFILES[id] ? { ...WEATHER_PROFILES[id] } : null;
  }

  function setWeather(id, options = {}) {
    if (!Object.hasOwn(WEATHER_PROFILES, id)) {
      return {
        ok: false,
        message: `Unknown weather state: ${id}`,
        validStates: [...WEATHER_NAMES]
      };
    }

    if (transitionDuration > 0 &&
        weatherTransitioning &&
        transitionElapsed < transitionDuration) {
      fromParameters = { ...currentParameters };
    } else {
      fromParameters = { ...currentParameters };
    }

    const oldState = currentWeatherId;
    currentWeatherId = id;
    targetParameters = { ...WEATHER_PROFILES[id] };

    transitionElapsed = 0;
    transitionDuration = clamp(
      finite(options.transitionSeconds, 45),
      0,
      1800
    );

    weatherTransitioning = transitionDuration > 0;

    if (!weatherTransitioning) {
      currentParameters = { ...targetParameters };
      globalWetness = Math.max(
        globalWetness,
        targetParameters.surfaceWetness
      );
    }

    events.emit(WEATHER_EVENTS.changed, {
      from: oldState,
      to: id,
      transitionSeconds: transitionDuration,
      reason: options.reason ?? "scripted"
    });

    state.patch({ weatherId: id }, "weather-state");
    runtime.notify?.(
      `Weather: ${id}`,
      id === "hurricane" ? "error" : "info"
    );

    if (id === "hurricane") {
      events.emit("weather:stormWarning", {
        type: "hurricane",
        severity: "extreme",
        timestamp: new Date().toISOString()
      });
    }

    return { ok: true, from: oldState, to: id };
  }

  function registerWeatherZone(options) {
    const id = `weather-zone-${nextWeatherZoneId++}`;

    if (!Number.isFinite(options?.x) ||
        !Number.isFinite(options?.z) ||
        !Number.isFinite(options?.radius) ||
        options.radius <= 0) {
      throw new TypeError(
        "Weather zones require finite x, z and positive radius values."
      );
    }

    if (options.weatherId && !WEATHER_PROFILES[options.weatherId]) {
      throw new RangeError(`Unknown zone weather: ${options.weatherId}`);
    }

    const zone = {
      id,
      x: options.x,
      z: options.z,
      radius: options.radius,
      weatherId: options.weatherId ?? null,
      windMultiplier: clamp(finite(options.windMultiplier, 1), 0, 3),
      precipitationMultiplier: clamp(
        finite(options.precipitationMultiplier, 1), 0, 3
      ),
      expiresAt: Number.isFinite(options.durationSeconds)
        ? elapsedWeatherTime + Math.max(0, options.durationSeconds)
        : Infinity
    };

    weatherZones.set(id, zone);

    return Object.freeze({
      id,
      dispose: () => weatherZones.delete(id)
    });
  }

  function getZoneAt(position) {
    if (!position) return null;

    let best = null;
    let bestInfluence = 0;

    for (const zone of weatherZones.values()) {
      const distance = Math.hypot(
        position.x - zone.x,
        position.z - zone.z
      );

      if (distance >= zone.radius) continue;

      const influence = 1 - distance / zone.radius;

      if (influence > bestInfluence) {
        bestInfluence = influence;
        best = { zone, influence };
      }
    }

    return best;
  }

  function getWindAt(position, time = elapsedWeatherTime) {
    const x = finite(position?.x, 0);
    const z = finite(position?.z, 0);
    const localWeather = getWeatherAt(position);

    const baseAngle =
      finite(localWeather.windDirectionRad, 0.68) +
      Math.sin(time * 0.008) * 0.22 +
      (smoothNoise2D(x * 0.0011, z * 0.0011, 112) - 0.5) * 0.3;

    const gustNoise = smoothNoise2D(
      x * 0.012 + time * 0.11,
      z * 0.012 - time * 0.07,
      1997
    );

    const gust = lerp(0.78, 1.28, gustNoise);
    const zoneInfo = getZoneAt(position);
    const localMultiplier = zoneInfo
      ? lerp(1, zoneInfo.zone.windMultiplier, zoneInfo.influence)
      : 1;

    let speed = Math.max(0, localWeather.windSpeed) * gust * localMultiplier;

    if (currentWeatherId === "hurricane") {
      const distanceFactor = Math.min(1, Math.hypot(x, z) / 1200);
      speed *= 0.8 + distanceFactor * 0.3;
    }

    return new B.Vector3(
      Math.cos(baseAngle) * speed,
      Math.sin(time * 0.35 + x * 0.005) * speed * 0.035,
      Math.sin(baseAngle) * speed
    );
  }

  function calculateWindForce(body, position) {
    const mass = finite(body?.massKg, NaN);
    const dragCoefficient = clamp(
      finite(body?.dragCoefficient, 1),
      0,
      4
    );
    const area = clamp(finite(body?.areaM2, 1), 0.01, 200);

    if (!Number.isFinite(mass) || mass <= 0) {
      throw new TypeError("Wind-force bodies require massKg > 0.");
    }

    const velocity = body?.velocity;
    const bodyVelocity = velocity
      ? new B.Vector3(
          finite(velocity.x),
          finite(velocity.y),
          finite(velocity.z)
        )
      : B.Vector3.Zero();

    const relativeWind = getWindAt(position).subtract(bodyVelocity);
    const relativeSpeed = relativeWind.length();

    const rho = 1.225;
    const forceMagnitude =
      0.5 * rho * dragCoefficient * area *
      relativeSpeed * relativeSpeed;

    const force = relativeSpeed > 1e-5
      ? relativeWind.scale(forceMagnitude / relativeSpeed)
      : B.Vector3.Zero();

    const maxForce = mass * 30;

    if (force.length() > maxForce) {
      force.normalize().scaleInPlace(maxForce);
    }

    return force;
  }

  function getWeatherAt(position) {
    const zoneInfo = getZoneAt(position);
    const profile = { ...currentParameters };

    if (zoneInfo?.zone.weatherId) {
      const zoneProfile = WEATHER_PROFILES[zoneInfo.zone.weatherId];
      const t = clamp(zoneInfo.influence, 0, 1);

      for (const key of Object.keys(profile)) {
        profile[key] = lerp(profile[key], zoneProfile[key], t);
      }
    }

    const localNoise = smoothNoise2D(
      finite(position?.x) * 0.0015,
      finite(position?.z) * 0.0015,
      221
    );

    profile.windSpeed *= 0.9 + localNoise * 0.2;

    if (zoneInfo) {
      profile.precipitation *= lerp(
        1,
        zoneInfo.zone.precipitationMultiplier,
        zoneInfo.influence
      );
    }

    return profile;
  }

  function getSurfaceEffectsAt(position) {
    const localWeather = getWeatherAt(position);

    const tractionMultiplier = clamp(
      1 - globalWetness * 0.28 -
      localWeather.precipitation * 0.08,
      0.48,
      1
    );

    const brakeMultiplier = clamp(
      1 - globalWetness * 0.14,
      0.62,
      1
    );

    return {
      wetness: globalWetness,
      tractionMultiplier,
      brakeMultiplier,
      visibilityM: localWeather.visibilityM,
      wind: getWindAt(position),
      precipitation: localWeather.precipitation
    };
  }

  function setTimeOfDay(minutes, options = {}) {
    const next = ((finite(minutes) % 1440) + 1440) % 1440;
    timeMinutes = next;
    lastPublishedMinute = Math.floor(next);

    state.patch({ timeOfDayMinutes: next }, "time-of-day");

    events.emit(WEATHER_EVENTS.timeChanged, {
      minutes: next,
      formatted: formatClock(next),
      source: options.source ?? "scripted"
    });

    return { ok: true, minutes: next };
  }

  function formatClock(minutes) {
    const normalized = ((minutes % 1440) + 1440) % 1440;

    return (
      `${String(Math.floor(normalized / 60)).padStart(2, "0")}:` +
      `${String(Math.floor(normalized % 60)).padStart(2, "0")}`
    );
  }

  function setSeasonDay(day) {
    seasonDay = clamp(Math.floor(day), 1, 365);
    return { ok: true, dayOfYear: seasonDay };
  }

  function applyClouds() {
    for (const layer of cloudLayers) {
      layer.setCoverage(currentParameters.cloudCover);
      layer.setOpacity?.(currentParameters.cloudOpacity);
    }
  }

  function setRainEmission() {
    if (!rainSystem) return;

    rainSystem.emitRate = Math.round(
      3200 * clamp(currentParameters.precipitation, 0, 1)
    );

    if (currentParameters.precipitation < 0.02) {
      rainSystem.stop();
    } else if (!rainSystem.isStarted()) {
      rainSystem.start();
    }
  }

  function sampleWetness(dt) {
    const deltaSeconds = clamp(finite(dt), 0, 0.1);
    const wetnessTarget = clamp(currentParameters.surfaceWetness, 0, 1);

    if (currentParameters.precipitation > 0.02) {
      // Integrate wetting in simulated seconds. Rain moves surfaces toward
      // the active profile's wetness target instead of pushing every storm
      // indefinitely toward fully saturated surfaces.
      const rainRate = 0.035 + currentParameters.precipitation * 0.12;

      if (globalWetness < wetnessTarget) {
        globalWetness = Math.min(
          wetnessTarget,
          globalWetness + rainRate * deltaSeconds
        );
      } else if (globalWetness > wetnessTarget) {
        // When transitioning from a more severe storm to lighter rain, avoid
        // an instantaneous jump in friction; relax toward the new target.
        globalWetness = Math.max(
          wetnessTarget,
          globalWetness - 0.025 * deltaSeconds
        );
      }
    } else {
      const daylight =
        smoothstep01(clamp((solar.altitude + 0.1) / 0.65, 0, 1));
      const evaporationRate = 0.00055 + daylight * 0.0014;
      globalWetness = Math.max(0, globalWetness - evaporationRate * deltaSeconds);
    }

    globalWetness = clamp(globalWetness, 0, 1);
  }

  function emitEnvironmentImpact() {
    const second = Math.floor(elapsedWeatherTime);

    if (second === lastWeatherEventSecond) return;
    if (second < 1) return;

    lastWeatherEventSecond = second;

    events.emit(WEATHER_EVENTS.impact, {
      weatherId: currentWeatherId,
      parameters: { ...currentParameters },
      wetness: globalWetness,
      windSpeed: currentParameters.windSpeed,
      visibilityM: currentParameters.visibilityM,
      timestamp: elapsedWeatherTime
    });
  }

  async function initialize() {
    if (initialized) return { ok: true };

    initialized = true;

    fromParameters = { ...WEATHER_PROFILES.sunny };
    currentParameters = { ...WEATHER_PROFILES.sunny };
    targetParameters = { ...WEATHER_PROFILES.sunny };

    updateSolarModel();
    applyFog();
    applyClouds();

    const weatherService = {
      getWeatherId: () => currentWeatherId,
      getParameters,
      getWeatherProfile,
      setWeather,
      getWeatherAt,
      getWindAt,
      calculateWindForce,
      getSurfaceEffectsAt,
      getWetness: () => globalWetness,
      registerWeatherReactiveMaterial(material) {
        if (!material || typeof material !== "object") {
          throw new TypeError("Weather-reactive material must be an object.");
        }
        sceneMaterials.add(material);
        return () => sceneMaterials.delete(material);
      },
      registerWeatherZone,
      setTimeOfDay,
      getTimeOfDay: () => timeMinutes,
      getSunAngles,
      setSeasonDay,
      getSeasonDay: () => seasonDay
    };

    serviceDisposers.push(
      runtime.services.register("weather", weatherService)
    );

    serviceDisposers.push(
      runtime.services.register("environment", weatherService)
    );

    disposers.push(
      runtime.events.on("water:splash", payload => {
        if (!payload?.position) return;

        const splash = {
          position: payload.position,
          intensity: clamp(finite(payload.intensity, 0.2), 0, 1),
          time: elapsedWeatherTime
        };

        events.emit("weather:waterInteraction", splash);
      })
    );

    logger.info("Weather simulation initialized.", {
      latitudeDeg: tropicalLatitudeRad * 180 / Math.PI,
      dayOfYear: seasonDay
    });

    return { ok: true };
  }

  function start() {
    running = true;
    return { ok: true };
  }

  function update(context) {
    if (!initialized || !running || disposed) return;

    const dt = clamp(finite(context.gameDeltaSeconds), 0, 0.1);
    elapsedWeatherTime += dt;

    const previousMinute = lastPublishedMinute;
    timeMinutes = ((timeMinutes + dt * 0.8) % 1440 + 1440) % 1440;
    const currentMinute = Math.floor(timeMinutes);

    // Keep continuous astronomical time internally, but only publish state
    // changes at minute boundaries; this avoids forcing UI rerenders at 60 Hz.
    if (currentMinute !== previousMinute) {
      lastPublishedMinute = currentMinute;
      state.patch({ timeOfDayMinutes: currentMinute }, "weather-clock");
      events.emit(WEATHER_EVENTS.timeChanged, {
        minutes: timeMinutes,
        formatted: formatClock(timeMinutes),
        source: "simulation"
      });
    }

    if (weatherTransitioning) {
      transitionElapsed += dt;

      const t = smoothstep01(
        transitionDuration <= 0
          ? 1
          : transitionElapsed / transitionDuration
      );

      currentParameters = interpolateWeather(
        fromParameters,
        targetParameters,
        t
      );

      if (transitionElapsed >= transitionDuration) {
        currentParameters = { ...targetParameters };
        weatherTransitioning = false;
      }
    }

    if (elapsedWeatherTime - lastParameterEventTime >= 0.5) {
      lastParameterEventTime = elapsedWeatherTime;
      events.emit(WEATHER_EVENTS.blended, {
        weatherId: currentWeatherId,
        transitioning: weatherTransitioning,
        parameters: { ...currentParameters },
        wetness: globalWetness,
        timeOfDayMinutes: timeMinutes
      });
    }

    for (const [id, zone] of weatherZones) {
      if (elapsedWeatherTime >= zone.expiresAt) {
        weatherZones.delete(id);
      }
    }

    sampleWetness(dt);
    updateSolarModel();
    applyFog();
    applyClouds();
    setRainEmission();
    emitEnvironmentImpact();

    const center = camera.globalPosition || camera.position;

    stars.update(center, getNightFactor());

    for (const layer of cloudLayers) {
      layer.update(
        dt,
        currentParameters.windSpeed,
        center,
        currentParameters.windDirectionRad
      );
    }

    for (const material of sceneMaterials) {
      if (!material || material.isDisposed?.()) {
        sceneMaterials.delete(material);
        continue;
      }

      material.metadata = {
        ...(material.metadata || {}),
        weatherWetness: globalWetness
      };
      graphics?.setMaterialWetness?.(material, globalWetness);
    }

    if (elapsedWeatherTime - lastWindSampleTime >= 3) {
      lastWindSampleTime = elapsedWeatherTime;
      events.emit("weather:windSample", {
        wind: getWindAt(center),
        wetness: globalWetness
      });
    }

    state.patch({
      weatherId: currentWeatherId
    }, "weather-current-state");
  }

  function stop() {
    running = false;
    return { ok: true };
  }

  function dispose() {
    if (disposed) return;

    disposed = true;
    running = false;

    for (const off of disposers.splice(0)) {
      try { off(); } catch (error) {
        logger.warn("Weather event cleanup failed.", error);
      }
    }

    for (const off of serviceDisposers.splice(0)) {
      try { off(); } catch (error) {
        logger.warn("Weather service cleanup failed.", error);
      }
    }

    for (const cloud of cloudLayers) cloud.dispose();

    rainSystem?.dispose();
    rainTexture?.dispose();

    sky.dome.dispose();
    sky.material.dispose();

    stars.mesh.dispose();
    stars.mesh.material?.dispose();

    sun.dispose();
    sunMaterial.dispose();

    moon.dispose();
    moonMaterial.dispose();

    weatherZones.clear();
    sceneMaterials.clear();
  }

  return Object.freeze({
    initialize,
    start,
    update,
    stop,
    dispose,

    getWeatherId: () => currentWeatherId,
    getParameters,
    getWeatherAt,
    getWeatherProfile,
    setWeather,
    getWindAt,
    calculateWindForce,
    getSurfaceEffectsAt,
    registerWeatherZone,
    setTimeOfDay,
    getTimeOfDay: () => timeMinutes,
    getSunAngles,
    setSeasonDay,
    getSeasonDay: () => seasonDay
  });
}
