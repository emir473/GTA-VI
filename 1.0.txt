  <script>
    /*
     * Loads the pinned Babylon.js UMD build before importing the native
     * ES-module game runtime. Game modules never depend on script order
     * beyond this explicitly awaited library initialization.
     */
    (() => {
      const status = document.getElementById("loading-details");
      const message = document.getElementById("boot-message");

      window.__LEONIDA_BOOT_UI__ = {
        progress(percent, label, detail) {
          const p = Math.max(0, Math.min(100, Number(percent) || 0));
          document.getElementById("loading-bar").style.width = `${p}%`;
          document.getElementById("loading-percent").textContent =
            `${Math.round(p)}%`;
          document.getElementById("loading-progress")
            .setAttribute("aria-valuenow", String(Math.round(p)));

          if (label) {
            document.getElementById("loading-label").textContent = label;
          }
          if (detail) status.textContent = detail;
        },

        fail(error) {
          message.classList.add("error");
          message.textContent =
            error instanceof Error ? error.message : String(error);
        }
      };

      const library = document.createElement("script");
      library.src =
        "https://cdn.jsdelivr.net/npm/babylonjs@9.32.0/babylon.js";
      library.async = true;
      library.crossOrigin = "anonymous";

      library.onload = async () => {
        if (!window.BABYLON) {
          window.__LEONIDA_BOOT_UI__.fail(
            "Babylon.js was loaded but the global runtime is unavailable."
          );
          return;
        }

        try {
          await import("./loader.js");
        } catch (error) {
          console.error("Leonida module loader failed:", error);
          window.__LEONIDA_BOOT_UI__.fail(error);
        }
      };

      library.onerror = () => {
        window.__LEONIDA_BOOT_UI__.fail(
          "Could not download Babylon.js. Check the internet connection " +
          "or serve the pinned library locally."
        );
      };

      document.head.appendChild(library);
    })();
  </script>
